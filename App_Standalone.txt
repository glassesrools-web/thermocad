const gTBE = (zone, altitude) => {
  // Bp و Dp لهما نفس قيم B و D حسب الجدول 2.2 من DTR الرسمي
  let actualZone = zone;
  if (zone === "Bp") actualZone = "B";
  if (zone === "Dp") actualZone = "D";
  const alt = parseFloat(altitude) || 0;
  const table = TBE_T[actualZone];
  if (!table) return null;
  for (const [lo, hi, tbe] of table) {
    if (alt >= lo && alt < hi) return tbe;
  }
  return null;
};

// COMPUTE FUNCTION — CORRECTED per official DTR C3-2
// التعديلات:
// 1. تصحيح مناطق Bp و Dp
// 2. إضافة خيار "chauffe" للسقف والأرضية (مسخن - يدخل فقط في S1/S2 لـ Dref)
// 3. فصل مساحة الأرضية عن المحيط (S2 للأرضية على تربة)
// 4. تصحيح حساب cin
// 5. إضافة خيار K يدوي للأبواب والنوافذ
// 6. Dli = 0.2 × Ds (ثابت حسب طلبك)
// ═══════════════════════════════════════════════════════════════════
function compute(st) {
  // التحقق من صحة المدخلات
  const tbe = gTBE(st.zone, n(st.altitude));
  if (tbe === null) return { error: "Zone/altitude invalide" };
  const tbi = n(st.tbi) || 21;
  const dT = tbi - tbe;
  const Vh = n(st.Vh) || 100;
  const np = Math.min(parseInt(st.nb_pieces) || 1, 5);
  let Ds = 0, Dlnc = 0, S1 = 0, S2 = 0, S3 = 0, S4 = 0, S5 = 0;
  const det = [];
  const det_lnc = [];
  // ─────────────────────────────────────────────────────────────────
  // 1. جدران خارجية
  // ─────────────────────────────────────────────────────────────────
  for (const p of (st.parois_ext || [])) {
    const A = n(p.surface), R = n(p.R);
    if (!A || !R) continue;
    const rs = RS[`${p.orientation || "lateral"}_ext`] || 0.17;
    const K = f4(1 / (rs + R));
    const ds = f4(K * A);
    Ds += ds;
    S3 += A;
    det.push({ nom: p.nom || "Mur extérieur", A, K, tau: "—", ds });
  }
  // ─────────────────────────────────────────────────────────────────
  // 2. جدران LNC (على محل غير مدفأ)
  // ─────────────────────────────────────────────────────────────────
  for (const p of (st.parois_lnc || [])) {
    const A = n(p.surface), R = n(p.R);
    if (!A || !R) continue;
    const tau = getTau(p, 0.5);
    const rs = RS[`${p.orientation || "lateral"}_lnc`] || 0.22;
    const K = f4(1 / (rs + R));
    const ds = f4(tau * K * A);
    Dlnc += ds;
    det_lnc.push({ nom: p.nom || "Mur LNC", A, K, tau, ds, type: "wall_lnc" });
  }
  // ─────────────────────────────────────────────────────────────────
  // 3. السقف (S1) — من الغرف، كل غرفة لها خصائص مستقلة
  // ─────────────────────────────────────────────────────────────────
  for (const tor of (st.toitures_rooms || [])) {
    const A = n(tor.surface);
    if (!A) continue;
    const contact = tor.contact || "ext";
    if (contact === "chauffe") {
      det.push({ nom: "Toiture (chauffée — non incluse dans Dref)", A, K: 0, ds: 0, tau: "—", note: "paroi intérieure" });
    } else {
      const R = n(tor.R);
      if (R > 0) {
        const rs = RS[`ascendant_${contact === "lnc" ? "lnc" : "ext"}`] || 0.14;
        const K = f4(1 / (rs + R));
        let tau = 1, ds = 0;
        if (contact === "lnc") {
          tau = getTau(tor, 0.9); ds = f4(tau * K * A);
          Dlnc += ds;
          S1 += A;
          det_lnc.push({ nom: `Toiture (sur LNC) — ${tor.roomName || ""}`, A, K, tau, ds, type: "roof_lnc" });
        } else {
          ds = f4(K * A);
          Ds += ds;
          S1 += A;
          det.push({ nom: `Toiture (extérieure) — ${tor.roomName || ""}`, A, K, tau: "—", ds });
        }
      }
    }
  }
  // ─────────────────────────────────────────────────────────────────
  // 4. أرضية LNC (على محل غير مدفأ) — من الغرف
  // ─────────────────────────────────────────────────────────────────
  for (const rm of (st.planchers_lnc_rooms || [])) {
    const pl = rm.plancher || ROOM_THERMAL_DEFAULTS.plancher;
    const A = n(rm.area || 0);
    const R = n(pl.R);
    if (!A || !R) continue;
    const tau = getTau(pl, 0.65);
    const K = f4(1 / (RS.descendant_lnc + R));
    const ds = f4(tau * K * A);
    Dlnc += ds;
    S2 += A;
    det_lnc.push({ nom: `Plancher LNC — ${rm.name || rm.id}`, A, K, tau, ds, type: "floor_lnc" });
  }
  // ─────────────────────────────────────────────────────────────────
  // 5. أرضية مسخنة — من الغرف (جدار داخلي — لا تدخل في Dref ولا Dsol)
  // ─────────────────────────────────────────────────────────────────
  for (const rm of (st.planchers_chauf_rooms || [])) {
    const A = n(rm.area || 0);
    if (!A) continue;
    // S2 لا تُضاف — الأرضية المسخنة جدار داخلي مثل السقف المسخن
    det.push({ nom: `Plancher chauffé — ${rm.name || rm.id}`, A, K: 0, ds: 0, tau: "—", note: "paroi intérieure" });
  }
  // ─────────────────────────────────────────────────────────────────
  // 6. نوافذ (S5) - مع إمكانية إدخال K يدوي
  // ─────────────────────────────────────────────────────────────────
  for (const f of (st.fenetres || [])) {
    const A = n(f.surface);
    if (!A) continue;
    let K;
    if (f.k_manuel && n(f.k_manuel) > 0) {
      K = f4(n(f.k_manuel));  // استخدام K يدوي إذا تم إدخاله
    } else {
      K = gKV(f.type || "double", n(f.lame_mm) || 12, f.cadre || "bois");
    }
    if (f.contact === "lnc") {
      // نافذة على جدار LNC → تُضاف إلى Dlnc مع معامل τ الخاص بها
      const tau = getTau(f, 0.5);
      const ds = f4(tau * K * A);
      Dlnc += ds;
      S5 += A;
      det_lnc.push({ nom: f.nom || "Fenêtre LNC", A, K, tau, ds, type: "window_lnc", k_manuel: f.k_manuel || "auto" });
    } else {
      const ds = f4(K * A);
      Ds += ds;
      S5 += A;
      det.push({ nom: f.nom || "Fenêtre", A, K, tau: "—", ds, k_manuel: f.k_manuel || "auto" });
    }
  }
  // ─────────────────────────────────────────────────────────────────
  // 7. أبواب (S4) - مع إمكانية إدخال K يدوي
  // ─────────────────────────────────────────────────────────────────
  for (const p of (st.portes || [])) {
    const A = n(p.surface);
    if (!A) continue;
    let K;
    if (p.k_manuel && n(p.k_manuel) > 0) {
      K = f4(n(p.k_manuel));  // استخدام K يدوي إذا تم إدخاله
    } else {
      K = KP_T[`${p.materiau || "bois"}_${p.type_vitrage || "opaque"}_${p.contact || "ext"}`] || 3.5;
    }
    if (p.contact === "lnc") {
      // باب على جدار LNC → يُضاف إلى Dlnc مع معامل τ الخاص بالجدار
      const tau = getTau(p, 0.5);
      const ds = f4(tau * K * A);
      Dlnc += ds;
      S4 += A;
      det_lnc.push({ nom: p.nom || "Porte LNC", A, K, tau, ds, type: "door_lnc", k_manuel: p.k_manuel || "auto" });
    } else {
      const ds = f4(K * A);
      Ds += ds;
      S4 += A;
      det.push({ nom: p.nom || "Porte", A, K, tau: "—", ds, k_manuel: p.k_manuel || "auto" });
    }
  }
  Ds = f4(Ds);
  Dlnc = f4(Dlnc);
  // ─────────────────────────────────────────────────────────────────
  // 8. الجسور الحرارية Dli = 20% من Ds فقط (لا يشمل Dlnc)
  // ─────────────────────────────────────────────────────────────────
  const Dli = f4(0.20 * Ds);
  // ─────────────────────────────────────────────────────────────────
  // 9. فقدان التربة Dsol — من الغرف (type=sol) + جدران مدفونة
  // ─────────────────────────────────────────────────────────────────
  let Dsol = 0;
  const det_sol = [];
  // 9a. أرضيات على تربة من الغرف
  for (const rm of (st.planchers_sol_rooms || [])) {
    const pl = rm.plancher || {};
    // المحيط التلقائي من نقاط الغرفة (÷50 لتحويل البكسل إلى متر) — يُستخدم إذا لم يُدخَل يدوياً
    const autoPerim = rm.points && rm.points.length > 1
      ? parseFloat((rm.points.reduce((s, p, i) => {
          const nx = rm.points[(i + 1) % rm.points.length];
          return s + Math.hypot(nx.x - p.x, nx.y - p.y);
        }, 0) / 50).toFixed(2))
      : 0;
    const perim = n(pl.perimetre) || autoPerim;
    const zs = parseFloat(pl.z) || 0;
    const A = n(rm.area || 0);
    if (!perim || zs === undefined) continue;
    const typeIso = pl.type_iso || "sans_iso";
    const rIso = n(pl.r_iso) || 1.0;
    const ks = f4(gKSol(zs, typeIso, rIso));
    const ds_sol = f4(ks * perim);
    Dsol += ds_sol;
    if (A > 0) S2 += A;
    det_sol.push({ nom: `Plancher/Sol — ${rm.name || rm.id}`, ks, p: perim, surface: A, ds: ds_sol, type: typeIso });
  }
  // 9b. جدران مدفونة (من الغرف + من الرسم)
  for (const m of (st.murs_enterres || [])) {
    const p_m = n(m.perimetre), R_m = n(m.R), z_m = parseFloat(m.z) || 0;
    if (!p_m || !R_m) continue;
    const K_m = f4(1 / (RS.lateral_ext + R_m));
    const ks_m = f4(gKS58(z_m, K_m));
    const ds_m = f4(ks_m * p_m);
    Dsol += ds_m;
    det_sol.push({ nom: m.nom || "Mur enterré", ks: ks_m, K: K_m, p: p_m, ds: ds_m, type: "mur_enterre" });
  }
  Dsol = f4(Dsol);
  // ─────────────────────────────────────────────────────────────────
  // 10. DT الكلي
  // ─────────────────────────────────────────────────────────────────
  const DT = f4(Ds + Dli + Dlnc + Dsol);
  // ─────────────────────────────────────────────────────────────────
  // 11. Dref (المرجع) - حسب الجدول 2.1 من DTR
  // ─────────────────────────────────────────────────────────────────
  const cf = (DREF_T[st.type_log] || DREF_T.collectif)[st.zone] || [1, 2.4, 1.2, 3.5, 4.5];
  const Dref = f4(cf[0] * S1 + cf[1] * S2 + cf[2] * S3 + cf[3] * S4 + cf[4] * S5);
  const reg_ok = DT <= +(1.05 * Dref).toFixed(4);
  // ─────────────────────────────────────────────────────────────────
  // 12. تجديد الهواء Qv (الفصل 7)
  // ─────────────────────────────────────────────────────────────────
  const nsdb = parseInt(st.nb_sdb) || 1;
  const nae = parseInt(st.nb_autre_eau) || 0;
  const nwc = parseInt(st.nb_wc) || 1;
  const Qvmin_v = np < 5 ? (QVMIN[np] || 100) : 110 + (Math.max(parseInt(st.nb_pieces) || 1, 5) - 5) * 10;
  const Qvmax_v = (QVMAX_CUI[np] || 135) + (QVMAX_SDB[np] || 30) * nsdb + 15 * nae + (QVMAX_WC[np] || 30) * nwc;
  const Qvref = f2((5 * Qvmin_v + Qvmax_v) / 6);
  const Qv = f2(Math.max(0.6 * Vh, Qvref));
  // ─────────────────────────────────────────────────────────────────
  // 13. تسرب الريح Qs (الفصل 7)
  // ─────────────────────────────────────────────────────────────────
  let Qs = 0;
  for (const o of (st.ouvrants_vent || [])) {
    const A = n(o.surface);
    if (!A) continue;
    Qs += (PO_T[o.type] || 4.0) * A * Math.sqrt(gEV(n(o.H) || 4, o.rugosite || "IV"));
  }
  Qs = f2(Qs / 2);
  const DR = f4(0.34 * (Qv + Qs));
  const DRv = f4(0.34 * Qv);
  const DRs = f4(0.34 * Qs);
  // ─────────────────────────────────────────────────────────────────
  // 14. معاملات التصحيح cr و cin
  // ─────────────────────────────────────────────────────────────────
  const CR = { individuel: 0, central_tout_isole: 0.05, central_partiel: 0.10, central_non_isole: 0.20 };
  const cr = CR[st.type_chauf] ?? 0.10;
  // ⭐ تصحيح حساب cin (إصلاح الخطأ)
  let cin = 0.10;  // قيمة افتراضية للتشغيل المستمر
  if (st.mode_chauf === "discontinu") {
    if (st.inertie === "forte") cin = 0.20;
    else cin = 0.15;  // moyenne أو faible
  } else {
    cin = 0.10;  // continu
  }
  // ─────────────────────────────────────────────────────────────────
  // 15. الاستطاعة الحرارية النهائية Q - المعادلة (2.9) من DTR
  // ─────────────────────────────────────────────────────────────────
  const Q = +((tbi - tbe) * ((1 + Math.max(cr, cin)) * DT + (1 + cr) * DR)).toFixed(1);
  const DB = +((DT + DR) * dT).toFixed(1);
  // ─────────────────────────────────────────────────────────────────
  // 16. النتائج
  // ─────────────────────────────────────────────────────────────────
  return {
    tbe, tbi, dT: f2(dT),
    det, det_lnc, det_sol,
    Ds, Dli, Dsol, Dlnc, DT, Dref, reg_ok,
    Qv, Qvref, Qvmin: Qvmin_v, Qvmax: Qvmax_v,
    Qs, DR, DRv, DRs,
    cr, cin,
    Q, Qkw: +(Q / 1000).toFixed(3), DB,
    S1, S2, S3, S4, S5
  };
}

  const l2 = dx * dx + dy * dy;
  if (!l2) return { pt: { ...a }, t: 0 };
  const t = Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / l2));
  return { pt: { x: a.x + t * dx, y: a.y + t * dy }, t };
};
const centroid = (pts) => {
  // FIX: prevent division by zero if pts is empty or undefined
  if (!pts || pts.length === 0) return { x: 0, y: 0 };
  return {
    x: pts.reduce((s, p) => s + p.x, 0) / pts.length,
    y: pts.reduce((s, p) => s + p.y, 0) / pts.length
  };
};
const roundRect = (ctx, x, y, w, h, r) => {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.arcTo(x + w, y, x + w, y + r, r);
  ctx.lineTo(x + w, y + h - r);
  ctx.arcTo(x + w, y + h, x + w - r, y + h, r);
  ctx.lineTo(x + r, y + h);
  ctx.arcTo(x, y + h, x, y + h - r, r);
  ctx.lineTo(x, y + r);
  ctx.arcTo(x, y, x + r, y, r);
  ctx.closePath();
};
const wallLengthM = w => Math.hypot(w.x2 - w.x1, w.y2 - w.y1) / SC;
const roomHsl = i => `hsl(${(i * 67 + 30) % 360}, 58%, 55%)`;
const roomHsla = (i, a) => `hsla(${(i * 67 + 30) % 360}, 58%, 55%, ${a})`;
const pointInPolygon = (pt, polygon) => {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i].x, yi = polygon[i].y;
    const xj = polygon[j].x, yj = polygon[j].y;
    if (((yi > pt.y) !== (yj > pt.y)) && (pt.x < (xj - xi) * (pt.y - yi) / (yj - yi) + xi)) inside = !inside;
  }
  return inside;
};
const makeWall = (id, x1, y1, x2, y2, height = DEFAULT_H, extra = {}) => {
  const length = parseFloat(wallLengthM({ x1, y1, x2, y2 }).toFixed(3));
  const grossArea = parseFloat((length * height).toFixed(3));
  return {
    id, x1, y1, x2, y2, height, length, grossArea,
    wallType: "ext", manualType: false, R: 0.48, orientation: "lateral",
    tau: 0.5, tau_mode: "auto", type_lnc: "circ_indirecte_i_i",
    isBuried: false, buriedDepth: "-1.0", /* ⭐ جديد: دعم الجدار المدفون */
    ...extra
  };
};

// GEOMETRY — Segment intersection & wall splitting
// ═══════════════════════════════════════════════════════════════════
const segIntersect = (a1, a2, b1, b2) => {
  const dx1 = a2.x - a1.x, dy1 = a2.y - a1.y;
  const dx2 = b2.x - b1.x, dy2 = b2.y - b1.y;
  const denom = dx1 * dy2 - dy1 * dx2;
  if (Math.abs(denom) < 1e-8) return null;
  const t = ((b1.x - a1.x) * dy2 - (b1.y - a1.y) * dx2) / denom;
  const u = ((b1.x - a1.x) * dy1 - (b1.y - a1.y) * dx1) / denom;
  const EPS = 1e-4;
  if (t > EPS && t < 1 - EPS && u > EPS && u < 1 - EPS) {
    return { x: a1.x + t * dx1, y: a1.y + t * dy1, t, u };
  }
  return null;
};
const splitWallsAtIntersections = (existingWalls, newWall, nwidFn) => {
  const hits = [];
  for (const ew of existingWalls) {
    const ix = segIntersect(
      { x: newWall.x1, y: newWall.y1 }, { x: newWall.x2, y: newWall.y2 },
      { x: ew.x1, y: ew.y1 }, { x: ew.x2, y: ew.y2 }
    );
    if (ix) hits.push({ ...ix, ew });
  }
  if (hits.length === 0) {
    return { walls: [...existingWalls, newWall], newWallIds: [newWall.id] };
  }
  hits.sort((a, b) => a.t - b.t);
  const pts = [{ x: newWall.x1, y: newWall.y1 }, ...hits.map(h => ({ x: h.x, y: h.y })), { x: newWall.x2, y: newWall.y2 }];
  const newSegs = [];
  const dtBase = {
    wallType: newWall.wallType, manualType: newWall.manualType,
    R: newWall.R, orientation: newWall.orientation,
    tau: newWall.tau, tau_mode: newWall.tau_mode, type_lnc: newWall.type_lnc,
    isBuried: newWall.isBuried || false, buriedDepth: newWall.buriedDepth || "-1.0"
  };
  for (let i = 0; i < pts.length - 1; i++) {
    const p1 = pts[i], p2 = pts[i + 1];
    if (Math.hypot(p2.x - p1.x, p2.y - p1.y) < 2) continue;
    newSegs.push(makeWall(nwidFn(), p1.x, p1.y, p2.x, p2.y, newWall.height, dtBase));
  }
  let updWalls = [...existingWalls];
  for (const h of hits) {
    const idx = updWalls.findIndex(w => w.id === h.ew.id);
    if (idx === -1) continue;
    const ew = updWalls[idx];
    const sp = { x: h.x, y: h.y };
    if (Math.hypot(sp.x - ew.x1, sp.y - ew.y1) < 2 || Math.hypot(sp.x - ew.x2, sp.y - ew.y2) < 2) continue;
    const dtEw = {
      wallType: ew.wallType, manualType: ew.manualType,
      R: ew.R, orientation: ew.orientation,
      tau: ew.tau, tau_mode: ew.tau_mode, type_lnc: ew.type_lnc,
      isBuried: ew.isBuried || false, buriedDepth: ew.buriedDepth || "-1.0"
    };
    const wA = makeWall(nwidFn(), ew.x1, ew.y1, sp.x, sp.y, ew.height, dtEw);
    const wB = makeWall(nwidFn(), sp.x, sp.y, ew.x2, ew.y2, ew.height, dtEw);
    updWalls = [...updWalls.slice(0, idx), wA, wB, ...updWalls.slice(idx + 1)];
  }
  return { walls: [...updWalls, ...newSegs], newWallIds: newSegs.map(w => w.id) };
};

const detectRoomsFromGraph = (currentWalls, existingRooms, defaultH, rRefObj) => {
  if (currentWalls.length < 2) return existingRooms;
  const EPS = 4;
  const verts = [];
  const getVIdx = (x, y) => {
    for (let i = 0; i < verts.length; i++) {
      if (Math.abs(verts[i].x - x) < EPS && Math.abs(verts[i].y - y) < EPS) return i;
    }
    verts.push({ x, y });
    return verts.length - 1;
  };
  const halfEdges = [];
  currentWalls.forEach(w => {
    const a = getVIdx(w.x1, w.y1);
    const b = getVIdx(w.x2, w.y2);
    if (a === b) return;
    halfEdges.push({ from: a, to: b, wallId: w.id });
    halfEdges.push({ from: b, to: a, wallId: w.id });
  });
  if (!halfEdges.length) return existingRooms;
  const outEdges = verts.map(() => []);
  halfEdges.forEach((he, i) => {
    const fv = verts[he.from], tv = verts[he.to];
    outEdges[he.from].push({ heIdx: i, angle: Math.atan2(tv.y - fv.y, tv.x - fv.x), to: he.to });
  });
  outEdges.forEach(arr => arr.sort((a, b) => a.angle - b.angle));
  const nextHE = new Array(halfEdges.length).fill(-1);
  halfEdges.forEach((he, i) => {
    const arr = outEdges[he.to];
    const revPos = arr.findIndex(e => e.to === he.from);
    if (revPos === -1) return;
    const prevPos = (revPos - 1 + arr.length) % arr.length;
    nextHE[i] = arr[prevPos].heIdx;
  });
  const used = new Set();
  const faces = [];
  for (let start = 0; start < halfEdges.length; start++) {
    if (used.has(start) || nextHE[start] === -1) continue;
    const faceVerts = [], faceWallIds = new Set();
    let curr = start, iter = 0;
    while (!used.has(curr) && iter++ < halfEdges.length + 1) {
      used.add(curr);
      faceVerts.push(halfEdges[curr].from);
      faceWallIds.add(halfEdges[curr].wallId);
      const nxt = nextHE[curr];
      if (nxt === -1 || nxt === start) break;
      curr = nxt;
    }
    if (faceVerts.length < 3) continue;
    const pts = faceVerts.map(vi => ({ ...verts[vi] }));
    let area2 = 0;
    for (let j = 0; j < pts.length; j++) {
      const nx = pts[(j + 1) % pts.length];
      area2 += pts[j].x * nx.y - nx.x * pts[j].y;
    }
    // فقط الـ faces ذات المساحة الموجبة (CW في إحداثيات الشاشة)
    if (area2 > SC * SC * 0.01) {
      faces.push({ pts, areaM2: parseFloat((area2 / 2 / (SC * SC)).toFixed(3)), wallIds: [...faceWallIds] });
    }
  }
  const usedExisting = new Set();
  return faces.map(face => {
    let matched = null;
    for (const rm of existingRooms) {
      if (usedExisting.has(rm.id)) continue;
      if (!rm.points || rm.points.length < 3) continue;
      const rmC = centroid(rm.points);
      if (pointInPolygon(rmC, face.pts)) { matched = rm; break; }
    }
    if (!matched) {
      let bestScore = 0;
      for (const rm of existingRooms) {
        if (usedExisting.has(rm.id)) continue;
        const rmSet = new Set(rm.wallIds || []);
        const fSet = new Set(face.wallIds);
        let ov = 0;
        for (const id of fSet) if (rmSet.has(id)) ov++;
        const score = ov / Math.max(fSet.size, rmSet.size, 1);
        if (score > bestScore && score >= 0.5) { bestScore = score; matched = rm; }
      }
    }
    if (matched) {
      usedExisting.add(matched.id);
      return { ...matched, points: face.pts, area: face.areaM2, wallIds: face.wallIds };
    }
    const ci = ++rRefObj.current;
    return {
      id: `r${ci}`, name: `Pièce ${ci}`, points: face.pts, area: face.areaM2,
      colorIdx: ci, color: roomHsl(ci), roomHeight: defaultH, wallIds: face.wallIds,
      // ⭐ الخصائص الحرارية للسقف والأرضية لكل غرفة
      toiture:  { ...ROOM_THERMAL_DEFAULTS.toiture },
      plancher: { ...ROOM_THERMAL_DEFAULTS.plancher },
    };
  });
};

          type_lnc: wall?.type_lnc || "circ_indirecte_i_i"
        })
      };
    }),
  };
}

                          <div key={k} style={{ flex: 1, background: theme === "dark" ? "rgba(10,20,34,0.7)" : "rgba(255,255,255,0.25)", borderRadius: 4, padding: "3px 5px", textAlign: "center" }}>
                            <div style={{ fontSize: 7.5, color: theme === "dark" ? "#3a5570" : "#94a3b8" }}>{k}</div>
                            <div style={{ fontSize: 11, fontWeight: 800, color: c, fontFamily: "monospace" }}>{v}</div>
                          </div>
                        ))}
                      </div>
                    </div>
                    {/* الصيغة النهائية */}
                    <div style={{ background: theme === "dark" ? "rgba(10,30,16,0.8)" : "rgba(220,252,231,0.5)", border: "2px solid #166534", borderRadius: 8, padding: "9px 10px" }}>
                      <div style={{ fontWeight: 700, fontSize: 10, color: "#22c55e", marginBottom: 5 }}>{t("finalFormula")}</div>
                      <Formula text={`Q = ΔT × [(1+max(cr,cin))×DT + (1+cr)×DR]\n  = ${dtrRes.dT} × [(1+${maxCrCin.toFixed(2)})×${dtrRes.DT} + (1+${dtrRes.cr})×${dtrRes.DR}]\n  = ${dtrRes.Q.toLocaleString()} W`}/>
                      <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 1fr", gap: 3, marginBottom: 6 }}>
                        {[["ΔT", `${dtrRes.dT} °C`, "#fbbf24"], ["DT", `${dtrRes.DT} W/°C`, "#60a5fa"], ["DR", `${dtrRes.DR} W/°C`, "#f472b6"]].map(([k, v, c]) => (
                          <div key={k} style={{ background: theme === "dark" ? "rgba(6,18,8,0.6)" : "rgba(220,252,231,0.3)", borderRadius: 4, padding: "3px 5px", textAlign: "center" }}>
                            <div style={{ fontSize: 7.5, color: "#3a5570" }}>{k}</div>
                            <div style={{ fontSize: 9, fontWeight: 800, color: c, fontFamily: "monospace" }}>{v}</div>
                          </div>
                        ))}
                      </div>
                      <div style={{ textAlign: "center", fontSize: 30, fontWeight: 900, color: "#22c55e", padding: "8px 0", background: theme === "dark" ? "rgba(6,18,8,0.8)" : "rgba(220,252,231,0.4)", borderRadius: 6 }}>
                        {dtrRes.Qkw} <span style={{ fontSize: 14 }}>kW</span>
                        <div style={{ fontSize: 11, opacity: 0.7, fontWeight: 600 }}>{dtrRes.Q.toLocaleString()} W</div>
                      </div>
                    </div>
                  </div>
                  );
                })())}
              </div>
            </>
          ) : (
            <div style={{ flex: 1, display: "flex", alignItems: "center", justifyContent: "center", flexDirection: "column", gap: 8 }}>
              <div style={{ fontSize: 32, opacity: 0.15 }}>🔥</div>
              <div style={{ fontSize: 10, color: theme === "dark" ? "#1a2a3a" : "#cbd5e1", textAlign: "center", padding: "0 16px" }}>{t("tabDTR")}</div>
            </div>
          )}
        </div>

      </div>
    </div>
  );
}
