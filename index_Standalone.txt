import React from 'react';
import { createRoot } from 'react-dom/client';
import App from './App';

// ─── Styles globaux (complètent ceux de public/index.html) ─────────
const globalStyle = document.createElement('style');
globalStyle.textContent = `
  /* Sélection de texte */
  ::selection        { background: rgba(34,197,94,0.25); color: inherit; }
  ::-moz-selection   { background: rgba(34,197,94,0.25); color: inherit; }

  /* Focus visible amélioré */
  :focus-visible {
    outline: 2px solid #22c55e;
    outline-offset: 2px;
    border-radius: 3px;
  }

  /* Transitions douces sur les éléments interactifs */
  button, select, input[type="range"] {
    transition: background 0.15s, border-color 0.15s, box-shadow 0.15s;
  }

  /* Inputs number — masque les flèches par défaut */
  input[type="number"]::-webkit-inner-spin-button,
  input[type="number"]::-webkit-outer-spin-button {
    -webkit-appearance: none;
    margin: 0;
  }
  input[type="number"] { -moz-appearance: textfield; }

  /* Curseur de redimensionnement sur canvas (modifiable via JS) */
  canvas[data-mode="edit"]   { cursor: default !important; }
  canvas[data-mode="wall"]   { cursor: crosshair !important; }
  canvas[data-mode="door"]   { cursor: cell !important; }
  canvas[data-mode="window"] { cursor: cell !important; }
`;
document.head.appendChild(globalStyle);

// ─── Montage React ─────────────────────────────────────────────────
const container = document.getElementById('root');

if (!container) {
  throw new Error('[ThermoCAD] #root element not found in index.html');
}

const root = createRoot(container);

root.render(
  // StrictMode désactivé car html2canvas + canvas ne s'entendent pas
  // bien avec le double-rendu du mode strict en développement.
  // Réactivez-le si vous n'utilisez pas html2canvas directement.
  <App />
);
